     1→   353→## Cover Page Design System
     2→   354→
     3→   355→### Design Philosophy
     4→   356→
     5→   357→Covers use **7 validated layout recipes** + **parameterized variants** instead of free combination. Each recipe's background, layout, and decoration are visually verified. Differentiation comes from **palette × font size × content variation**.
     6→   358→
     7→   359→**Architecture principle:** All recipes use a **single 16838 outer wrapper table** (one row, exact height). Recipes R1/R2/R3 use **ZERO nested tables** — all decoration is achieved via **paragraph borders** (left/right/top/bottom). This ensures maximum cross-engine stability (MS Office + WPS).
     8→   360→
     9→   361→### Cover Color Palettes (Dark Mode + Light Mode)
    10→   362→
    11→   363→Each palette defines 3 core colors (Background, Primary, Accent) + derived cover/table tokens.
    12→   364→
    13→   365→⚠️ **Disambiguation: `cover.titleColor` is a COLOR value, not title text.** All keys under `cover: { ... }` are **color hex codes** for styling cover text elements. They are NOT text content. The actual title text comes from `config.title`. Never use `P.titleColor` as the `text` parameter of a `TextRun` — it must only be used as the `color` parameter.
    14→   366→
    15→   367→```js
    16→   368→// ✅ Correct — config.title is the text, P.titleColor is the color
    17→   369→new TextRun({ text: config.title, color: P.titleColor })
    18→   370→
    19→   371→// ❌ WRONG — using color value as text content
    20→   372→new TextRun({ text: P.titleColor, color: P.titleColor })  // displays "FFFFFF" as visible text!
    21→   373→```

   115→   467→### ⚠️ Dark Cover → Light Table Rule
   116→   468→
   117→   469→Covers with dark backgrounds (DS-1, IG-1, DM-1, FG-1, GO-1) use bright accent on dark bg.

   119→   471→High-saturation accent colors (DM-1 #37DCF2, FG-1 #3DDBB5, SN-2 #B13DC6) are explicitly overridden in `table.*` fields above.

   149→   501→### ⚠️ Recipe Routing Rules (Replaces Free Selection)
   150→   502→
   151→   503→```js
   152→   504→function selectCoverRecipe(docType, industry, titleLength) {
   153→   505→  // No cover for these types
   154→   506→  if (["contract", "official", "exam", "resume"].includes(docType)) return null;
   155→   507→
   156→   508→  // Academic
   157→   509→  if (docType === "academic") return { recipe: "R5", palette: "ACADEMIC" };
   158→   510→
   159→   511→  // Thesis proposal report (开题报告)
   160→   512→  if (docType === "proposal_report") return { recipe: "R5", palette: "ACADEMIC" };
   161→   513→
   162→   514→  // Lesson plans — R6 editorial for arts/general, R4 for STEM
   163→   515→  if (docType === "lesson_plan" || docType === "lessonplan") {
   164→   516→    const stemKeywords = ["math", "physics", "chemistry", "biology", "science", "tech", "computer", "engineering"];
   165→   517→    if (stemKeywords.some(k => (industry || "").toLowerCase().includes(k))) {
   166→   518→      return { recipe: "R4", palette: "DM-1" };
   167→   519→    }
   168→   520→    // Arts, general, and all other lesson plans → R6 editorial
   169→   521→    return { recipe: "R6", palette: "ED-1" };
   170→   522→  }
   171→   523→
   172→   524→  // Creative/branding/design (formal) → R3 centered card frame
   173→   525→  if (["creative", "branding", "design"].includes(docType)) {
   174→   526→    return { recipe: "R3", palette: "SN-2" };
   175→   527→  }
   176→   528→
   177→   529→  // Cultural/newsletter/internal (casual) → R6 editorial
   178→   530→  if (["cultural", "newsletter", "internal"].includes(docType)) {
   179→   531→    return { recipe: "R6", palette: "ED-1" };
   180→   532→  }
   181→   533→
   182→   534→  // Activity/event planning → R6 editorial
   183→   535→  if (docType === "activity") return { recipe: "R6", palette: "ED-1" };
   184→   536→
   185→   537→  // Trend/research reports in cultural/creative/brand fields → R7 Swiss Tech
   186→   538→  if (docType === "trend_report" || docType === "research_report") {
   187→   539→    if (["cultural", "creative", "brand", "design"].includes(industry)) {
   188→   540→      return { recipe: "R7", palette: "ST-1" };
   189→   541→    }
   190→   542→  }
   191→   543→
   192→   544→  // Formal/business subtypes
   193→   545→  if (docType === "whitepaper") return { recipe: "R2", palette: industry === "finance" ? "IG-1" : "CM-2" };
   194→   546→  if (docType === "consulting") return { recipe: "R2", palette: "MIN-1" };
   195→   547→  if (docType === "proposal" || docType === "plan") return { recipe: "R4", palette: "GO-1" };
   196→   548→
   197→   549→  // Reports — palette by industry, use R1
   198→   550→  if (docType === "report") {
   199→   551→    const paletteMap = {
   200→   552→      finance: "IG-1", consulting: "MIN-1",
   201→   553→      tech: "DM-1", ai: "DM-1",
   202→   554→      education: "WM-1", green: "FG-1",
   203→   555→      medical: "MC-1", government: "GV-1",
   204→   556→    };
   205→   557→    return { recipe: "R1", palette: paletteMap[industry] || "DS-1" };
   206→   558→  }
   207→   559→
   208→   560→  // Default
   209→   561→  return { recipe: "R1", palette: "DS-1" };
   210→   562→}
   211→   563→
   212→   564→// ── Long-title override (applied AFTER initial recipe selection) ──
   213→   565→// Call this after selectCoverRecipe() when the actual title text is known.
   214→   566→function applyLongTitleOverride(result, titleLength) {
   215→   567→  if (!result || !result.recipe) return result;
   216→   568→  // R5 (academic) is never overridden — it has its own calcTitleLayoutMixed()
   217→   569→  if (result.recipe === "R5") return result;
   218→   570→  // R6 (editorial) is designed for short titles only (≤20 chars, single line)
   219→   571→  // Long titles → fall back to R1 (handles long titles best)
   220→   572→  if (titleLength > 20 && result.recipe === "R6") {
   221→   573→    return { recipe: "R1", palette: "WM-1" }; // ED-1 has no dark bg, use WM-1 (warm teal)
   222→   574→  }
   223→   575→  // R3/R4 struggle with long titles → fall back to R1 (same palette)
   224→   576→  if (titleLength > 20 && ["R3", "R4"].includes(result.recipe)) {
   225→   577→    return { recipe: "R1", palette: result.palette };
   226→   578→  }
   227→   579→  // Very long titles: even R2 centered looks scattered → R1 left-aligned
   228→   580→  if (titleLength > 30 && result.recipe === "R2") {
   229→   581→    return { recipe: "R1", palette: result.palette };
   230→   582→  }
   231→   583→  return result;
   232→   584→}
   233→   585→```

   262→   614→### Cover Title Length Guidelines
   263→   615→
   264→   616→When the user does NOT specify an exact title, the model should craft a title within the recommended range. If the user provides a title that exceeds the comfortable range, apply long-title routing in `selectCoverRecipe()` (see above).
   265→   617→
   266→   618→| Recipe | Comfortable (1–2 lines) | Maximum (3 lines) | Long Title Tolerance |
   267→   619→|--------|------------------------|--------------------|----------------------|
   268→   620→| R1     | 8–20 chars             | ≤50 chars          | ⭐⭐⭐⭐⭐ Best (left-aligned, full-page bg) |
   269→   621→| R2     | 8–18 chars             | ≤45 chars          | ⭐⭐⭐⭐ Good (full-page bg, but centered) |
   270→   622→| R3     | 8–15 chars             | ≤40 chars          | ⭐⭐ Poor (narrowest width) |
   271→   623→| R4     | 8–18 chars             | ≤45 chars          | ⭐⭐ Poor (fixed-height color block) |
   272→   624→| R5     | 8–16 chars             | ≤42 chars          | ⭐⭐⭐ OK (academic only, mixed-width calc) |
   273→   625→| R6     | 8–15 chars             | ≤20 chars (1 line) | ⭐ Single line only (editorial, no multi-line) |
   274→   626→| R7     | 8–24 chars             | ≤42 chars (3 lines) | ⭐⭐⭐⭐ Good (left-aligned, light bg, dynamic spacing) |
   275→   627→
   276→   628→**Title crafting rules (when model generates the title):**
   277→   629→1. Prefer concise titles within the "comfortable" range
   278→   630→2. If topic requires detail, split into title + subtitle (e.g., title="数字化转型战略研究" subtitle="——以某某企业为例")
   279→   631→3. Never exceed the "maximum" range unless user explicitly provides the full title

   283→   635→### ⚠️ Cover Page Break Rules
   284→   636→
   285→   637→Cover should be an independent section — **no PageBreak at the end needed**. The next section automatically starts a new page.
   286→   638→
   287→   639→```js
   288→   640→// ✅ Correct — cover is a separate section, no trailing PageBreak
   289→   641→sections: [
   290→   642→  { properties: { /* Cover section, margin all 0 */ }, children: buildCover(...) },
   291→   643→  { properties: { /* Body section */ }, children: buildContent(...) },
   292→   644→]
   293→   645→```

   295→   647→### ⚠️ Cover Content Overflow Prevention (Mandatory)
   296→   648→
   297→   649→1. Cover section page margin is 0; total content height ≤ 15638 twips (1200 twips safety margin for cross-engine compatibility — MS Office renders large fonts taller than calculated).
   298→   650→2. Color block Table `height` must use `rule: "exact"` (never `"atLeast"`).
   299→   651→3. Each recipe code includes height budget annotations — verify during generation.
   300→   652→4. **`verticalAlign` must always be `"top"`**. Never use `"center"` or `"bottom"` in exact-height rows — content will be clipped or overflow. Use `spacing.before` on the first paragraph for vertical positioning.
   301→   653→5. **Title font size MUST be dynamically calculated via `calcTitleLayout()`** (see below). Never hardcode font sizes above 40pt for cover titles. Every recipe MUST call `calcTitleLayout()` before building title paragraphs.
   302→   654→6. **Never use `margins.top`/`margins.bottom` for vertical positioning inside exact-height cells**: Cell margins reduce available height unpredictably across MS Office and WPS. Use `spacing.before` on the first paragraph instead. Only `margins.left`/`margins.right` are safe.
   303→   655→7. **Dynamic spacing is mandatory**: Use `calcCoverSpacing()` to compute `spacing.before` values dynamically based on content element count and title line count. Never use fixed large spacing values (e.g., `before: 4500`) that assume a specific title length.
   304→   656→8. **Cover must be a single-page section**: The cover section must produce exactly ONE page. If content overflows to a second page, it means the height budget is violated. Common overflow causes and fixes:
   305→   657→   - Title font too large → use `calcTitleLayout()` to auto-reduce
   306→   658→   - Too many meta lines → reduce font size or remove less important lines
   307→   659→   - Fixed `spacing.before` values too large → use `calcCoverSpacing()` for dynamic values

   366→   718→### ⚠️ Cover Title Layout — calcTitleLayout() (Mandatory for ALL Recipes)
   367→   719→
   368→   720→**Every cover recipe MUST use `calcTitleLayout()` to determine title font size and line breaks.** Hardcoding font sizes or passing the full title as a single TextRun is FORBIDDEN.
   369→   721→
   370→   722→**Every paragraph with font size > body text MUST set explicit line spacing** to prevent top clipping:
   371→   723→```js
   372→   724→// ★ MANDATORY: prevent inherited small line spacing from clipping large fonts
   373→   725→spacing: { line: Math.ceil(titlePt * 23), lineRule: "atLeast", after: 100 }
   374→   726→// Example: 36pt → line: 828; 44pt → line: 1012
   375→   727→```
   376→   728→Without this, the paragraph inherits body text line spacing (e.g., 560tw), which is shorter than the font height → top of characters gets clipped.
   377→   729→
   378→   730→```js
   379→   731→/**
   380→   732→ * Calculate safe font size and smart line breaks for cover titles.
   381→   733→ * MUST be called by every recipe before building title paragraphs.
   382→   734→ *
   383→   735→ * @param {string}  title          Full title string
   384→   736→ * @param {number}  maxWidthTwips  Available width for title text (twips, after subtracting margins/padding)
   385→   737→ * @param {number}  preferredPt    Desired max font size in pt (default 40)
   386→   738→ * @param {number}  minPt          Minimum allowed font size in pt (default 24)
   387→   739→ * @returns {{ titlePt: number, titleLines: string[] }}
   388→   740→ */
   389→   741→function calcTitleLayout(title, maxWidthTwips, preferredPt = 40, minPt = 24) {
   390→   742→  // Each CJK character width ≈ pt × 20 twips
   391→   743→  const charWidth = (pt) => pt * 20;
   392→   744→  const charsPerLine = (pt) => Math.floor(maxWidthTwips / charWidth(pt));
   393→   745→
   394→   746→  // Try from preferredPt downward until title fits in ≤ 3 lines
   395→   747→  let titlePt = preferredPt;
   396→   748→  let lines;
   397→   749→  while (titlePt >= minPt) {
   398→   750→    const cpl = charsPerLine(titlePt);
   399→   751→    if (cpl < 2) { titlePt -= 2; continue; }
   400→   752→    lines = splitTitleLines(title, cpl);
   401→   753→    if (lines.length <= 3) break;
   402→   754→    titlePt -= 2;
   403→   755→  }
   404→   756→
   405→   757→  // If still > 3 lines at minPt, force 3 lines
   406→   758→  if (!lines || lines.length > 3) {
   407→   759→    const cpl = charsPerLine(minPt);
   408→   760→    lines = splitTitleLines(title, cpl);
   409→   761→    titlePt = minPt;
   410→   762→  }
   411→   763→
   412→   764→  return { titlePt, titleLines: lines };
   413→   765→}
   414→   766→
   415→   767→/**
   416→   768→ * Smart Chinese title line-breaking — breaks at semantic boundaries, never mid-word.
   417→   769→ *
   418→   770→ * Rules:
   419→   771→ * 1. Prefer breaking after particles, punctuation, connectors, underscores, spaces
   420→   772→ * 2. Never split a compound word (e.g., "管理规范" must not become "管理规" + "范")
   421→   773→ * 3. No single-character orphan on the last line — merge into previous line
   422→   774→ * 4. If no good break point found within 60-130% of charsPerLine, break at charsPerLine
   423→   775→ *
   424→   776→ * @param {string} title        Full title string
   425→   777→ * @param {number} charsPerLine Max characters per line at current font size
   426→   778→ * @returns {string[]}          Array of line strings
   427→   779→ */
   428→   780→function splitTitleLines(title, charsPerLine) {
   429→   781→  if (title.length <= charsPerLine) return [title];
   430→   782→
   431→   783→  // Characters that are safe break points (break AFTER these)
   432→   784→  const breakAfter = new Set([
   433→   785→    ...'，。、；：！？',              // CJK punctuation
   434→   786→    ...'的与和及之在于为',            // CJK particles/prepositions
   435→   787→    ...'-_—–·/',                     // connectors
   436→   788→    ...' \t',                         // whitespace
   437→   789→  ]);
   438→   790→
   439→   791→  const lines = [];
   440→   792→  let remaining = title;
   441→   793→
   442→   794→  while (remaining.length > charsPerLine) {
   443→   795→    let breakAt = -1;
   444→   796→
   445→   797→    // Search backward from charsPerLine to 60% for a break point
   446→   798→    for (let i = charsPerLine; i >= Math.floor(charsPerLine * 0.6); i--) {
   447→   799→      if (i < remaining.length && breakAfter.has(remaining[i - 1])) {
   448→   800→        breakAt = i;
   449→   801→        break;
   450→   802→      }
   451→   803→    }
   452→   804→
   453→   805→    // If not found, search forward up to 130%
   454→   806→    if (breakAt === -1) {
   455→   807→      const limit = Math.min(remaining.length, Math.ceil(charsPerLine * 1.3));
   456→   808→      for (let i = charsPerLine + 1; i < limit; i++) {
   457→   809→        if (breakAfter.has(remaining[i - 1])) {
   458→   810→          breakAt = i;
   459→   811→          break;
   460→   812→        }
   461→   813→      }
   462→   814→    }
   463→   815→
   464→   816→    // Last resort: break at charsPerLine, but avoid splitting compound CJK words
   465→   817→    if (breakAt === -1) {
   466→   818→      breakAt = charsPerLine;
   467→   819→      // If both chars at the break boundary are CJK (likely a compound word),
   468→   820→      // step back 1 char to keep the word together
   469→   821→      const prevChar = remaining[breakAt - 1];
   470→   822→      const nextChar = remaining[breakAt];
   471→   823→      if (prevChar && nextChar &&
   472→   824→          !breakAfter.has(prevChar) && !breakAfter.has(nextChar) &&
   473→   825→          /[\u4e00-\u9fff]/.test(prevChar) && /[\u4e00-\u9fff]/.test(nextChar)) {
   474→   826→        breakAt = breakAt - 1;
   475→   827→      }
   476→   828→    }
   477→   829→
   478→   830→    lines.push(remaining.slice(0, breakAt).trim());
   479→   831→    remaining = remaining.slice(breakAt).trim();
   480→   832→  }
   481→   833→  if (remaining) lines.push(remaining);
   482→   834→
   483→   835→  // Prevent single-character orphan on last line — merge into previous
   484→   836→  if (lines.length > 1 && lines[lines.length - 1].length <= 2) {
   485→   837→    const last = lines.pop();
   486→   838→    lines[lines.length - 1] += last;
   487→   839→  }
   488→   840→
   489→   841→  return lines;
   490→   842→}
   491→   843→
   492→   844→/**
   493→   845→ * Calculate dynamic spacing values for cover elements to fit within page height.
   494→   846→ *
   495→   847→ * @param {object} params
   496→   848→ * @param {number} params.titleLineCount   Number of title lines
   497→   849→ * @param {number} params.titlePt          Title font size in pt
   498→   850→ * @param {boolean} params.hasSubtitle     Whether subtitle exists
   499→   851→ * @param {boolean} params.hasEnglishLabel Whether English label exists
   500→   852→ * @param {number} params.metaLineCount    Number of meta info lines
   501→   853→ * @param {number} params.fixedHeight      Sum of fixed-height elements (color strips, accent bars, footer) in twips
   502→   854→ * @param {number} params.pageHeight       Total page height in twips (default 16838)
   503→   855→ * @returns {{ topSpacing, midSpacing, bottomSpacing }}  Spacing values in twips
   504→   856→ */
   505→   857→function calcCoverSpacing(params) {
   506→   858→  const {
   507→   859→    titleLineCount = 1, titlePt = 36, hasSubtitle = false,
   508→   860→    hasEnglishLabel = false, metaLineCount = 0,
   509→   861→    fixedHeight = 800, pageHeight = 16838,
   510→   862→    marginTop = 0, marginBottom = 0,   // ★ NEW: pass actual section margins
   511→   863→  } = params;
   512→   864→
   513→   865→  // ★ Safety margin: 1200 twips (cross-engine: MS Office renders large fonts
   514→   866→  // taller than calculated; extra 400tw buffer prevents footer clipping)
   515→   867→  const SAFETY = 1200;
   516→   868→  // ★ Subtract page margins from available height (cover section may have margins)
   517→   869→  const usableHeight = pageHeight - marginTop - marginBottom - SAFETY;
   518→   870→
   519→   871→  // ★ Accurate height estimation per element:
   520→   872→  const titleHeight = titleLineCount * (titlePt * 23 + 200);
   521→   873→  const subtitleHeight = hasSubtitle ? (12 * 23 + 600) : 0;
   522→   874→  const englishLabelHeight = hasEnglishLabel ? (9 * 23 + 600) : 0;
   523→   875→  const metaHeight = metaLineCount * (10 * 23 + 100);
   524→   876→
   525→   877→  // ★ Account for implicit paragraph heights:
   526→   878→  const implicitParaHeight = 3 * 300;
   527→   879→
   528→   880→  const contentHeight = titleHeight + subtitleHeight + englishLabelHeight +
   529→   881→                        metaHeight + fixedHeight + implicitParaHeight;
   530→   882→
   531→   883→  const remainingSpace = usableHeight - contentHeight;
   532→   884→  const safeRemaining = Math.max(remainingSpace, 400);
   533→   885→
   534→   886→  // ★ Footer protection: bottomSpacing must be ≥ FOOTER_MIN to prevent
   535→   887→  // footer + accent line from being clipped at the cell bottom edge
   536→   888→  const FOOTER_MIN = 800;
   537→   889→  const rawTop = Math.floor(safeRemaining * 0.45);
   538→   890→  const rawBottom = Math.floor(safeRemaining * 0.45);
   539→   891→  const bottomSpacing = Math.max(rawBottom, FOOTER_MIN);
   540→   892→  const topSpacing = Math.max(rawTop - Math.max(0, FOOTER_MIN - rawBottom), 400);
   541→   893→  const midSpacing = Math.max(safeRemaining - topSpacing - bottomSpacing, 0);
   542→   894→
   543→   895→  return { topSpacing, midSpacing, bottomSpacing };
   544→   896→}
   545→   897→```
